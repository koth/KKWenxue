# JetPack 6.2 images commonly ship CMake 3.22. The project only uses
# FetchContent features available there; keep the native build usable without
# replacing the system toolchain.
cmake_minimum_required(VERSION 3.22)
project(audio8_native VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
set(PCRE2_BUILD_PCRE2_8 ON CACHE BOOL "" FORCE)
set(PCRE2_BUILD_PCRE2_16 OFF CACHE BOOL "" FORCE)
set(PCRE2_BUILD_PCRE2_32 OFF CACHE BOOL "" FORCE)
set(PCRE2_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(PCRE2_BUILD_PCRE2GREP OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(UTF8PROC_ENABLE_TESTING OFF CACHE BOOL "" FORCE)
FetchContent_Declare(json URL https://codeload.github.com/nlohmann/json/tar.gz/refs/tags/v3.11.3 URL_HASH SHA256=0d8ef5af7f9794e3263480193c491549b2ba6cc74bb018906202ada498a79406)
FetchContent_Declare(pcre2 URL https://github.com/PCRE2Project/pcre2/releases/download/pcre2-10.45/pcre2-10.45.tar.gz URL_HASH SHA256=0e138387df7835d7403b8351e2226c1377da804e0737db0e071b48f07c9d12ee)
FetchContent_Declare(utf8proc URL https://codeload.github.com/JuliaStrings/utf8proc/tar.gz/refs/tags/v2.9.0 URL_HASH SHA256=18c1626e9fc5a2e192311e36b3010bfc698078f692888940f1fa150547abb0c1)
FetchContent_MakeAvailable(json pcre2 utf8proc)
set(AUDIO8_TORCH_ROOT "" CACHE PATH "Explicit LibTorch root (Windows wheel development builds)")
if(WIN32 AND AUDIO8_TORCH_ROOT)
  find_package(CUDAToolkit REQUIRED)
  set(TORCH_INCLUDE_DIRS "${AUDIO8_TORCH_ROOT}/include" "${AUDIO8_TORCH_ROOT}/include/torch/csrc/api/include")
  foreach(lib torch torch_cpu torch_cuda c10 c10_cuda)
    list(APPEND TORCH_LIBRARIES "${AUDIO8_TORCH_ROOT}/lib/${lib}.lib")
  endforeach()
else()
  find_package(Torch REQUIRED)
  find_package(CUDAToolkit REQUIRED)
endif()
add_library(audio8_core src/io.cpp src/tokenizer.cpp src/model.cpp src/codec.cpp src/engine.cpp)
option(AUDIO8_CUDA_KERNELS "Build fused decode kernels" ON)
if(AUDIO8_CUDA_KERNELS)
  enable_language(CUDA)
  target_sources(audio8_core PRIVATE src/kernels.cu)
  target_compile_definitions(audio8_core PUBLIC AUDIO8_CUDA_KERNELS)
  set_target_properties(audio8_core PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
  # RoPE and SwiGLU keep the original intermediate rounding; do not fuse FMAs.
  target_compile_options(audio8_core PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false>)
  if(MSVC)
    target_compile_options(audio8_core PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-Xcompiler=/utf-8;--diag-suppress=1388,1390,1394>)
  endif()
endif()
target_include_directories(audio8_core PUBLIC include ${TORCH_INCLUDE_DIRS})
target_link_libraries(audio8_core PUBLIC ${TORCH_LIBRARIES} CUDA::cudart nlohmann_json::nlohmann_json pcre2-8-static utf8proc)
target_compile_definitions(audio8_core PRIVATE PCRE2_STATIC UTF8PROC_STATIC PCRE2_CODE_UNIT_WIDTH=8)
if(MSVC)
  target_compile_options(audio8_core PUBLIC $<$<COMPILE_LANGUAGE:CXX>:/utf-8;/bigobj;/EHsc;/Zc:__cplusplus>)
  target_compile_definitions(audio8_core PUBLIC NOMINMAX)
else()
  separate_arguments(TORCH_CXX_FLAGS_LIST NATIVE_COMMAND "${TORCH_CXX_FLAGS}")
  target_compile_options(audio8_core PUBLIC ${TORCH_CXX_FLAGS_LIST})
endif()
add_executable(audio8-tts src/main.cpp)
target_link_libraries(audio8-tts PRIVATE audio8_core)
if(AUDIO8_CUDA_KERNELS)
  add_executable(audio8-kernel-check EXCLUDE_FROM_ALL tests/check_kernels.cpp)
  target_link_libraries(audio8-kernel-check PRIVATE audio8_core)
endif()
install(TARGETS audio8-tts RUNTIME DESTINATION bin)
